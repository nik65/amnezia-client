cmake_minimum_required(VERSION 3.16)
project(HeadlessJournal LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Linux POSIX journal only")
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)
add_library(headless_journal journal.cpp)
set_target_properties(headless_journal PROPERTIES AUTOMOC OFF)
target_link_libraries(headless_journal PUBLIC Qt6::Core)
target_include_directories(headless_journal PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(headless_journal PUBLIC cxx_std_17)
target_compile_options(headless_journal PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(amnezia-headless-journal main.cpp)
set_target_properties(amnezia-headless-journal PROPERTIES AUTOMOC OFF)
target_link_libraries(amnezia-headless-journal PRIVATE headless_journal)
include(CTest)
if(BUILD_TESTING)
  add_executable(headless_journal_tests tests.cpp)
  set_target_properties(headless_journal_tests PROPERTIES AUTOMOC OFF)
  target_link_libraries(headless_journal_tests PRIVATE headless_journal)
  target_compile_options(headless_journal_tests PRIVATE -UNDEBUG)
  target_compile_definitions(headless_journal_tests PRIVATE JOURNAL_REFERENCE_TEST="${CMAKE_CURRENT_SOURCE_DIR}/python_reference_test.py" INSTALLER_REFERENCE="${CMAKE_CURRENT_SOURCE_DIR}/../../install_headless.sh")
  add_test(NAME journal_owned_fixtures COMMAND headless_journal_tests)
endif()
