cmake_minimum_required(VERSION 3.24)
project(amnezia_provisioning_verifier LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This verify-only producer implements Linux provisioning receipt checks")
endif()
set(OPENSSL_USE_STATIC_LIBS TRUE)
set(ZLIB_USE_STATIC_LIBS ON)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
add_executable(amnezia-verify-provisioning-native bootstrap_n.cpp)
target_compile_features(amnezia-verify-provisioning-native PRIVATE cxx_std_17)
target_compile_options(amnezia-verify-provisioning-native PRIVATE -Wall -Wextra)
target_link_options(amnezia-verify-provisioning-native PRIVATE -static)
target_link_libraries(amnezia-verify-provisioning-native PRIVATE OpenSSL::Crypto ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS})
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME provisioning_verifier_file_types
    COMMAND ${Python3_EXECUTABLE} -I -B ${CMAKE_CURRENT_SOURCE_DIR}/test_file_types.py
      $<TARGET_FILE:amnezia-verify-provisioning-native> ${CMAKE_CURRENT_BINARY_DIR}/test-attempts)
  set_tests_properties(provisioning_verifier_file_types PROPERTIES TIMEOUT 20)
endif()
