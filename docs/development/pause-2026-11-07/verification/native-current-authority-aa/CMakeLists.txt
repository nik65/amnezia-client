cmake_minimum_required(VERSION 3.16)
project(awg_native_current_authority_aa LANGUAGES CXX)
# AA lane only: frozen input snapshot in this directory, canonical tree untouched.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "native AWG control maintenance requires 64-bit Linux descriptor semantics")
endif()
set(AWG_DEPS_PREFIX "/home/ivano/amnezia-source-tests-20261104-native-deps-a/prefix" CACHE PATH "Verified json-c prefix")
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_path(JSON_C_INCLUDE_DIR NAMES json-c/json.h PATHS "${AWG_DEPS_PREFIX}/usr/include" NO_DEFAULT_PATH REQUIRED)
find_library(JSON_C_LIBRARY NAMES json-c PATHS "${AWG_DEPS_PREFIX}/usr/lib/x86_64-linux-gnu" NO_DEFAULT_PATH REQUIRED)
add_library(aa_txn_journal transaction_journal.cpp)
target_compile_features(aa_txn_journal PUBLIC cxx_std_17)
target_compile_options(aa_txn_journal PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(aa_txn_journal PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${JSON_C_INCLUDE_DIR}")
target_link_libraries(aa_txn_journal PUBLIC OpenSSL::Crypto "${JSON_C_LIBRARY}")
add_library(aa_txn_runtime maintenance_runtime.cpp)
target_compile_features(aa_txn_runtime PUBLIC cxx_std_17)
target_compile_options(aa_txn_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(aa_txn_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${JSON_C_INCLUDE_DIR}")
target_link_libraries(aa_txn_runtime PUBLIC aa_txn_journal OpenSSL::Crypto "${JSON_C_LIBRARY}")
enable_testing()
add_executable(aa_current_authority_tests test_current_authority.cpp)
target_compile_features(aa_current_authority_tests PRIVATE cxx_std_17)
target_compile_options(aa_current_authority_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
target_include_directories(aa_current_authority_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${JSON_C_INCLUDE_DIR}")
target_link_libraries(aa_current_authority_tests PRIVATE aa_txn_runtime OpenSSL::Crypto "${JSON_C_LIBRARY}")
add_test(NAME aa_current_authority_inert COMMAND aa_current_authority_tests)
set_tests_properties(aa_current_authority_inert PROPERTIES TIMEOUT 120)
