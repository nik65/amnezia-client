cmake_minimum_required(VERSION 3.25)
project(UpdateDownloadLifetimeFixture LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This inert fixture uses Linux /proc/self/fd for descriptor checks")
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Network)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(UPDATE_DOWNLOAD_FIXTURE_ASAN "Enable AddressSanitizer for the inert fixture" OFF)
set(updater_source "${CMAKE_CURRENT_SOURCE_DIR}/../../core/controllers/updateController.cpp")
set(generated_source "${CMAKE_CURRENT_BINARY_DIR}/fixture.cpp")
add_custom_command(OUTPUT "${generated_source}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_fixture.py"
        "${updater_source}" "${CMAKE_CURRENT_BINARY_DIR}" --generate-only
    DEPENDS "${updater_source}" "${CMAKE_CURRENT_SOURCE_DIR}/run_fixture.py"
    VERBATIM)
add_executable(update_download_lifetime_fixture "${generated_source}")
target_compile_features(update_download_lifetime_fixture PRIVATE cxx_std_17)
target_link_libraries(update_download_lifetime_fixture PRIVATE Qt6::Core Qt6::Network)
if(UPDATE_DOWNLOAD_FIXTURE_ASAN)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "ASAN option requires a GNU or Clang compiler")
    endif()
    target_compile_options(update_download_lifetime_fixture PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(update_download_lifetime_fixture PRIVATE -fsanitize=address)
endif()
include(CTest)
if(BUILD_TESTING)
    foreach(mode IN ITEMS legacy controller reply timer finished payload ready error hash size large)
        add_test(NAME update_download_${mode} COMMAND update_download_lifetime_fixture "${mode}")
        set_tests_properties(update_download_${mode} PROPERTIES TIMEOUT 10 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
    endforeach()
endif()
