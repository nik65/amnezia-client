cmake_minimum_required(VERSION 3.21)
project(LoggerConcurrencyContract LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This owned-directory test relies on Linux XDG isolation")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(Threads REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version.h" "#define APPLICATION_NAME \"LoggingConcurrencyFixture\"\n#define SERVICE_NAME \"LoggingConcurrencyService\"\n")
add_executable(logger_concurrency_test test.cpp "${REPO}/common/logger/logger.cpp" "${REPO}/common/logger/logger.h")
target_include_directories(logger_concurrency_test PRIVATE "${REPO}/common/logger" "${REPO}/client" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(logger_concurrency_test PRIVATE Qt6::Core Qt6::Gui Threads::Threads)
include(CTest)
add_test(NAME logger_concurrency COMMAND logger_concurrency_test)
set_tests_properties(logger_concurrency PROPERTIES TIMEOUT 30)
