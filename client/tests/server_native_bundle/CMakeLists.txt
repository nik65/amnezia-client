cmake_minimum_required(VERSION 3.25)
project(ServerNativeBundleTests LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(server_native_bundle_tests tst_server_native_bundle.cpp)
target_compile_features(server_native_bundle_tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(server_native_bundle_tests PRIVATE /W4 /WX)
else()
    target_compile_options(server_native_bundle_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
target_include_directories(server_native_bundle_tests PRIVATE ${CMAKE_CURRENT_LIST_DIR}/../..)
target_link_libraries(server_native_bundle_tests PRIVATE Qt6::Core)
enable_testing()
add_test(NAME server_native_bundle COMMAND server_native_bundle_tests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME server_native_bundle_cmake
    COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_LIST_DIR}/test_cmake_import.py
        --build-dir ${CMAKE_CURRENT_BINARY_DIR}/cmake-import-fixtures --cmake ${CMAKE_COMMAND})
set_tests_properties(server_native_bundle_cmake PROPERTIES TIMEOUT 60)
