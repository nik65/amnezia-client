cmake_minimum_required(VERSION 3.25)
project(HeadlessProvisioningContractTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
enable_testing()
add_executable(headless_provisioning_contract_tests tst_headless_provisioning_contract.cpp)
target_include_directories(headless_provisioning_contract_tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../..")
target_link_libraries(headless_provisioning_contract_tests PRIVATE Qt6::Core)
add_test(NAME headless_provisioning_contract COMMAND headless_provisioning_contract_tests)
