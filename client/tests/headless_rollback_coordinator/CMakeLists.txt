cmake_minimum_required(VERSION 3.25)
project(headless_rollback_coordinator_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC OFF)
include(CTest)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(actual_updater STATIC "${ROOT}/headless/headlessUpdateManager.cpp" runner_stub.cpp)
target_include_directories(actual_updater PUBLIC "${ROOT}/headless" "${ROOT}/client")
target_link_libraries(actual_updater PUBLIC Qt6::Core Qt6::Network OpenSSL::Crypto)
target_compile_options(actual_updater PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-missing-field-initializers)
add_executable(native_ack native_ack.cpp)
target_link_libraries(native_ack PRIVATE actual_updater)
target_compile_options(native_ack PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-missing-field-initializers -UNDEBUG)
add_test(NAME coordinator_actual_native_ack COMMAND native_ack)
add_library(actual_coordinator STATIC "${ROOT}/headless/rollbackCoordinator.cpp")
target_include_directories(actual_coordinator PUBLIC "${ROOT}/headless")
target_link_libraries(actual_coordinator PUBLIC actual_updater)
target_compile_options(actual_coordinator PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-missing-field-initializers)
add_executable(coordinator_fixtures coordinator.cpp)
target_link_libraries(coordinator_fixtures PRIVATE actual_coordinator)
target_compile_options(coordinator_fixtures PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-missing-field-initializers -UNDEBUG)
add_test(NAME coordinator_owned_phases COMMAND coordinator_fixtures)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory("${ROOT}/deploy/headless/native" native)
add_executable(actual_cli "${ROOT}/headless/cli.cpp" "${ROOT}/headless/headlessProtocol.cpp")
target_link_libraries(actual_cli PRIVATE actual_coordinator headless_fs_core Qt6::Core Qt6::Network)
target_compile_definitions(actual_cli PRIVATE AMNEZIA_HEADLESS_VERSION="5.0.3.14")
target_compile_options(actual_cli PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME coordinator_actual_cli_argv COMMAND ${Python3_EXECUTABLE} "${ROOT}/client/tests/headless_fs_cli/test_cli_fs.py" "$<TARGET_FILE:actual_cli>")
add_executable(profile_lock profile_lock.cpp "${ROOT}/headless/profileStore.cpp")
target_link_libraries(profile_lock PRIVATE actual_coordinator)
target_compile_options(profile_lock PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME coordinator_actual_profile_lock COMMAND profile_lock)
add_test(NAME coordinator_unprivileged_dispatch COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/test_dispatch.py" "$<TARGET_FILE:actual_cli>")
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(existing_updater "${ROOT}/headless/tests/tst_headless_updates.cpp")
set_target_properties(existing_updater PROPERTIES AUTOMOC ON)
target_link_libraries(existing_updater PRIVATE actual_updater Qt6::Test)
target_compile_definitions(existing_updater PRIVATE AMNEZIA_HEADLESS_VERSION="5.0.3.14")
add_test(NAME coordinator_existing_updater COMMAND existing_updater)
add_executable(existing_profiles "${ROOT}/headless/tests/tst_profile_store.cpp" "${ROOT}/headless/profileStore.cpp")
set_target_properties(existing_profiles PROPERTIES AUTOMOC ON)
target_include_directories(existing_profiles PRIVATE "${ROOT}/headless")
target_link_libraries(existing_profiles PRIVATE Qt6::Core Qt6::Network Qt6::Test)
add_test(NAME coordinator_existing_profiles COMMAND existing_profiles)
add_executable(existing_protocol "${ROOT}/headless/tests/tst_headless_protocol.cpp" "${ROOT}/headless/headlessProtocol.cpp")
set_target_properties(existing_protocol PROPERTIES AUTOMOC ON)
target_include_directories(existing_protocol PRIVATE "${ROOT}/headless")
target_link_libraries(existing_protocol PRIVATE Qt6::Core Qt6::Network Qt6::Test)
target_compile_definitions(existing_protocol PRIVATE HEADLESS_CLI_BINARY="$<TARGET_FILE:actual_cli>")
add_dependencies(existing_protocol actual_cli)
add_test(NAME coordinator_existing_protocol COMMAND existing_protocol)
add_executable(pending_response pending_response.cpp "${ROOT}/headless/headlessProtocol.cpp")
target_include_directories(pending_response PRIVATE "${ROOT}/headless")
target_link_libraries(pending_response PRIVATE Qt6::Core)
target_compile_definitions(pending_response PRIVATE ACTUAL_CLI_SOURCE="${ROOT}/headless/cli.cpp")
target_compile_options(pending_response PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME coordinator_pending_cli_response COMMAND pending_response)
