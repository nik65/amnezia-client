cmake_minimum_required(VERSION 3.25)
project(OwnerPublicationPermitTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(OpenSSL REQUIRED)
add_executable(owner_publication_permit_tests tst_owner_publication_permit.cpp)
target_include_directories(owner_publication_permit_tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../..")
target_link_libraries(owner_publication_permit_tests PRIVATE Qt6::Core OpenSSL::Crypto)
if(MSVC)
 target_compile_options(owner_publication_permit_tests PRIVATE /W4 /WX /permissive-)
else()
 target_compile_options(owner_publication_permit_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME owner_publication_permit COMMAND owner_publication_permit_tests)
