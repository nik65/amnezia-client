cmake_minimum_required(VERSION 3.25)
project(AndroidRestoredSessionTests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CLIENT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(REPOSITORY "${CLIENT_ROOT}/.." ABSOLUTE)
set(AUDIT_VPN_SOURCE "${CLIENT_ROOT}/vpnConnection.cpp" CACHE FILEPATH "Actual VPN callback source")
option(RESTORED_SESSION_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
 "${AUDIT_VPN_SOURCE}" "${CLIENT_ROOT}/core/controllers/connectionController.cpp"
 "${CLIENT_ROOT}/ui/controllers/connectionUiController.cpp" "${CMAKE_CURRENT_LIST_DIR}/generate_fixture.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/generate_fixture.py"
 "${REPOSITORY}" "${CMAKE_CURRENT_BINARY_DIR}/restored_fixture.cpp" --vpn-source "${AUDIT_VPN_SOURCE}" RESULT_VARIABLE GENERATED)
if(NOT GENERATED EQUAL 0)
 message(FATAL_ERROR "Actual restored-session source extraction failed")
endif()
add_executable(android_restored_session_tests "${CMAKE_CURRENT_BINARY_DIR}/restored_fixture.cpp"
 "${CLIENT_ROOT}/core/utils/errorCodes.h" "${CLIENT_ROOT}/core/utils/errorStrings.cpp")
target_include_directories(android_restored_session_tests PRIVATE "${CLIENT_ROOT}")
target_link_libraries(android_restored_session_tests PRIVATE Qt6::Core)
target_compile_features(android_restored_session_tests PRIVATE cxx_std_17)
if(NOT MSVC)
 target_compile_options(android_restored_session_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(RESTORED_SESSION_SANITIZERS)
  target_compile_options(android_restored_session_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(android_restored_session_tests PRIVATE -fsanitize=address,undefined)
 endif()
endif()
enable_testing()
foreach(CASE IN ITEMS restored-timeout refused-recovery refused-pending pending-confirmed confirmed wrong-nonce wrong-epoch failed-routes missing-owner reconnect-unconfirmed reconnect-confirmed)
 add_test(NAME android_restored_${CASE} COMMAND android_restored_session_tests "${CASE}")
 set_tests_properties(android_restored_${CASE} PROPERTIES TIMEOUT 10)
endforeach()
