cmake_minimum_required(VERSION 3.25)
project(NativeGatewayContextTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Qt6 REQUIRED COMPONENTS Core)
set(NATIVE_GATEWAY_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." CACHE PATH "Source tree whose actual methods are tested")
option(NATIVE_GATEWAY_REPRODUCE_BASELINE "Expect old startup/status mismatch using pre-fix source tree" OFF)
set(generator_args --source-root "${NATIVE_GATEWAY_SOURCE_ROOT}" --output "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(NATIVE_GATEWAY_REPRODUCE_BASELINE)
    list(APPEND generator_args --baseline)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_native_gateway_context.py" ${generator_args}
                RESULT_VARIABLE generation_result)
if(NOT generation_result EQUAL 0)
    message(FATAL_ERROR "Production source extraction/order validation failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${NATIVE_GATEWAY_SOURCE_ROOT}/client/vpnConnection.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_native_gateway_context.py")
add_executable(native_gateway_context_tests "${CMAKE_CURRENT_BINARY_DIR}/generated/test_gateway_context.cpp")
target_link_libraries(native_gateway_context_tests PRIVATE Qt6::Core)
target_compile_definitions(native_gateway_context_tests PRIVATE AMNEZIA_DESKTOP)
if(MSVC)
    target_compile_options(native_gateway_context_tests PRIVATE /UNDEBUG /Zc:__cplusplus /permissive-)
else()
    target_compile_options(native_gateway_context_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME native_gateway_context_lifecycle COMMAND native_gateway_context_tests)
