cmake_minimum_required(VERSION 3.21)
project(ApplicationLoggingCoverage LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version.h" "#define APPLICATION_NAME \"LoggingCoverageTest\"\n#define SERVICE_NAME \"LoggingCoverageService\"\n")
add_executable(test_logger test_logger.cpp "${REPO}/common/logger/logger.cpp" "${REPO}/common/logger/logger.h")
target_include_directories(test_logger PRIVATE "${REPO}/common/logger" "${REPO}/client" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(test_logger PRIVATE Qt6::Core Qt6::Gui)
