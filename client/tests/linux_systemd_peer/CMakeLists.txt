cmake_minimum_required(VERSION 3.25)
project(LinuxSystemdPeerTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
    message(FATAL_ERROR "Linux desktop only")
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Network Core5Compat)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(AMNEZIA_SYSTEMD REQUIRED IMPORTED_TARGET libsystemd>=246)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
add_executable(linux_systemd_peer_tests test.cpp "${ROOT}/ipc/linuxsystemdpeer.cpp")
target_include_directories(linux_systemd_peer_tests PRIVATE "${ROOT}/ipc")
target_link_libraries(linux_systemd_peer_tests PRIVATE Qt6::Core Qt6::Network PkgConfig::AMNEZIA_SYSTEMD Threads::Threads)
target_compile_options(linux_systemd_peer_tests PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(linux_systemd_peer_tests PRIVATE PACKAGED_UNIT_FIXTURE="${CMAKE_CURRENT_LIST_DIR}/fixtures/AmneziaVPN.service")
enable_testing()
add_test(NAME linux_systemd_peer COMMAND linux_systemd_peer_tests)
add_executable(linux_systemd_wire_tests wire_test.cpp)
target_include_directories(linux_systemd_wire_tests PRIVATE "${ROOT}")
target_link_libraries(linux_systemd_wire_tests PRIVATE Qt6::Core Qt6::Network PkgConfig::AMNEZIA_SYSTEMD Threads::Threads)
target_compile_options(linux_systemd_wire_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_compile_definitions(linux_systemd_wire_tests PRIVATE EXECSTART_ACCOUNTING_FIXTURE="${CMAKE_CURRENT_LIST_DIR}/fixtures/execstart-accounting-a14i3.json")
add_test(NAME linux_systemd_wire COMMAND linux_systemd_wire_tests)
add_executable(linux_systemd_transport_stage_tests transport_stage_test.cpp)
target_include_directories(linux_systemd_transport_stage_tests PRIVATE "${ROOT}/ipc")
target_link_libraries(linux_systemd_transport_stage_tests PRIVATE PkgConfig::AMNEZIA_SYSTEMD Threads::Threads)
target_compile_options(linux_systemd_transport_stage_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME linux_systemd_transport_stage COMMAND linux_systemd_transport_stage_tests)

# Compiled here but deliberately NOT registered as a test: running it requires
# a separately authorized owned guest and genuine installed root system service.
set(_source_hashes "{")
foreach(_path IN ITEMS ipc/localpeerauthentication.cpp ipc/localpeerauthentication.h
        ipc/linuxsystemdpeer.cpp ipc/linuxsystemdpeer.h ipc/linuxsystemdbustransport.h
        client/tests/linux_systemd_peer/runtime_verifier.cpp)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${_path}")
    file(SHA256 "${ROOT}/${_path}" _sha)
    string(APPEND _source_hashes "\"${_path}\":\"${_sha}\",")
endforeach()
string(REGEX REPLACE ",$" "}" _source_hashes "${_source_hashes}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/candidate_source_hashes.h"
    "inline constexpr char candidateSourceHashes[] = R\"AMNEZIA(${_source_hashes})AMNEZIA\";\n")
add_executable(linux_systemd_runtime_verifier runtime_verifier.cpp
    "${ROOT}/ipc/localpeerauthentication.cpp" "${ROOT}/ipc/linuxsystemdpeer.cpp")
target_include_directories(linux_systemd_runtime_verifier PRIVATE "${ROOT}/ipc" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(linux_systemd_runtime_verifier PRIVATE AMNEZIA_SYSTEMD_PEER_AUTH=1)
target_link_libraries(linux_systemd_runtime_verifier PRIVATE Qt6::Core Qt6::Core5Compat Qt6::Network PkgConfig::AMNEZIA_SYSTEMD)
target_compile_options(linux_systemd_runtime_verifier PRIVATE -Wall -Wextra -Werror)
