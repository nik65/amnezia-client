cmake_minimum_required(VERSION 3.25)
project(NativeSessionRestoreContract LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CLIENT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(WORKER_FIXTURE "${CMAKE_CURRENT_BINARY_DIR}/worker_fixture.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_worker_fixture.py"
    "${CLIENT_ROOT}" "${WORKER_FIXTURE}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    generate_worker_fixture.py ../../vpnConnection.cpp ../../core/protocols/androidVpnProtocol.cpp
    ../../core/controllers/connectionController.cpp ../../ui/controllers/connectionUiController.cpp
    ../../platforms/android/android_controller.cpp)
add_executable(native_session_restore_test test.cpp "${WORKER_FIXTURE}")
target_include_directories(native_session_restore_test PRIVATE "${CLIENT_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(native_session_restore_test PRIVATE cxx_std_17)
target_link_libraries(native_session_restore_test PRIVATE Qt6::Core)
if(MSVC)
    target_compile_options(native_session_restore_test PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(native_session_restore_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(RESTORE_SANITIZERS "Use address and undefined behavior sanitizers" OFF)
if(RESTORE_SANITIZERS AND NOT MSVC)
    target_compile_options(native_session_restore_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(native_session_restore_test PRIVATE -fsanitize=address,undefined)
endif()
include(CTest)
add_test(NAME native_session_restore COMMAND native_session_restore_test)
set_tests_properties(native_session_restore PROPERTIES TIMEOUT 30)
