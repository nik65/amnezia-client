cmake_minimum_required(VERSION 3.25)
project(AndroidDiscoveryWorkerTests LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(REPOSITORY "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${REPOSITORY}/client/vpnConnection.cpp" "${REPOSITORY}/client/core/controllers/connectionController.cpp" "${REPOSITORY}/client/ui/controllers/connectionUiController.cpp" "${CMAKE_CURRENT_LIST_DIR}/fixture.cpp.in" "${CMAKE_CURRENT_LIST_DIR}/generate_fixture.py")
set(AUDIT_WORKER_SOURCE "${REPOSITORY}/client/vpnConnection.cpp" CACHE FILEPATH "Actual worker source for before/after fixture")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/generate_fixture.py" "${REPOSITORY}" "${CMAKE_CURRENT_BINARY_DIR}/worker.cpp" "${AUDIT_WORKER_SOURCE}" RESULT_VARIABLE GENERATED)
if(NOT GENERATED EQUAL 0)
 message(FATAL_ERROR "Actual worker extraction failed")
endif()
add_executable(discovery_worker_tests "${CMAKE_CURRENT_BINARY_DIR}/worker.cpp")
target_include_directories(discovery_worker_tests PRIVATE "${REPOSITORY}/client")
target_link_libraries(discovery_worker_tests PRIVATE Qt6::Core)
target_compile_features(discovery_worker_tests PRIVATE cxx_std_17)
target_compile_options(discovery_worker_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME discovery_worker_contract COMMAND discovery_worker_tests)
set_tests_properties(discovery_worker_contract PROPERTIES TIMEOUT 30)

set(AUDIT_ENTRY_SOURCE "${REPOSITORY}/client/vpnConnection.cpp" CACHE FILEPATH "Actual entry source for before/after fixture")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/generate_entry_fixture.py" "${REPOSITORY}" "${CMAKE_CURRENT_BINARY_DIR}/entry.cpp" "${AUDIT_ENTRY_SOURCE}" RESULT_VARIABLE ENTRY_GENERATED)
if(NOT ENTRY_GENERATED EQUAL 0)
 message(FATAL_ERROR "Actual connect entry extraction failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/entry.cpp.in" "${CMAKE_CURRENT_LIST_DIR}/generate_entry_fixture.py")
add_executable(discovery_entry_tests "${CMAKE_CURRENT_BINARY_DIR}/entry.cpp")
target_include_directories(discovery_entry_tests PRIVATE "${REPOSITORY}/client")
target_link_libraries(discovery_entry_tests PRIVATE Qt6::Core)
target_compile_features(discovery_entry_tests PRIVATE cxx_std_17)
target_compile_options(discovery_entry_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME discovery_entry_contract COMMAND discovery_entry_tests)
set_tests_properties(discovery_entry_contract PROPERTIES TIMEOUT 10)
