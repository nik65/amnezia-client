cmake_minimum_required(VERSION 3.20)
project(AmneziaNativeLogCollector LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Collector is a Linux server executable; cross-build Linux, never deploy a host Windows binary")
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_library(collector_core STATIC collector_store.cpp)
target_link_libraries(collector_core PUBLIC OpenSSL::Crypto SQLite::SQLite3 Threads::Threads)
target_include_directories(collector_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(collector_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(amnezia-client-logs-collector collector_main.cpp collector_http.cpp)
target_link_libraries(amnezia-client-logs-collector PRIVATE collector_core)
target_compile_options(amnezia-client-logs-collector PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(collector_store_test collector_store_test.cpp)
  target_link_libraries(collector_store_test PRIVATE collector_core)
  add_test(NAME collector_store COMMAND collector_store_test)
  add_executable(collector_auth_fifo_test collector_auth_fifo_test.cpp)
  target_link_libraries(collector_auth_fifo_test PRIVATE collector_core)
  target_link_options(collector_auth_fifo_test PRIVATE "-Wl,--wrap=open")
  add_test(NAME collector_auth_fifo COMMAND collector_auth_fifo_test)
  set_tests_properties(collector_auth_fifo PROPERTIES TIMEOUT 5)
  add_executable(collector_http_test collector_http.cpp collector_http_test.cpp)
  target_link_libraries(collector_http_test PRIVATE Threads::Threads)
  add_test(NAME collector_http COMMAND collector_http_test)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME collector_network_blackbox
      COMMAND ${CMAKE_COMMAND} -E env AMNEZIA_NATIVE_COLLECTOR=$<TARGET_FILE:amnezia-client-logs-collector>
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_native_collector.py -v)
    add_test(NAME collector_memory_multiphase
      COMMAND ${CMAKE_COMMAND} -E env AMNEZIA_NATIVE_COLLECTOR=$<TARGET_FILE:amnezia-client-logs-collector>
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_native_collector_memory.py)
    set_tests_properties(collector_memory_multiphase PROPERTIES TIMEOUT 90 LABELS "memory;loopback")
  endif()
endif()
install(TARGETS amnezia-client-logs-collector RUNTIME DESTINATION bin)
