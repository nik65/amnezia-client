cmake_minimum_required(VERSION 3.25)
project(CollectorRouteNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT UNIX OR APPLE)
 message(FATAL_ERROR "Linux POSIX-only route reconciler")
endif()
find_path(JSON_C_INCLUDE_DIR NAMES json-c/json.h REQUIRED)
find_library(JSON_C_LIBRARY NAMES json-c REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(collector_route_core route.cpp posix.cpp)
target_include_directories(collector_route_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${JSON_C_INCLUDE_DIR})
target_link_libraries(collector_route_core PUBLIC ${JSON_C_LIBRARY} OpenSSL::Crypto)
target_compile_options(collector_route_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(amnezia-collector-route main.cpp)
target_link_libraries(amnezia-collector-route PRIVATE collector_route_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(collector_route_tests tests.cpp)
 target_link_libraries(collector_route_tests PRIVATE collector_route_core)
 target_compile_options(collector_route_tests PRIVATE -UNDEBUG)
 add_test(NAME collector_route_owned_fixture COMMAND collector_route_tests)
endif()
